\begin{frame}{Modello della rete}

\begin{itemize}
    \item La rete stradale $\mathcal{N}$ è rappresentata come un grafo orientato.
    \item Sono presenti due popolazioni:
    \begin{itemize}
        \item \textbf{hat}: origine $\widehat{\mathcal O}$, destinazione $\widehat{\mathcal D}$;
        \item \textbf{check}: origine $\widecheck{\mathcal O}$, destinazione $\widecheck{\mathcal D}$.
    \end{itemize}
    \item Ogni popolazione può scegliere tra diversi percorsi
    \[
    \widehat{\gamma}_1,\dots,\widehat{\gamma}_{\widehat n},
    \qquad
    \widecheck{\gamma}_1,\dots,\widecheck{\gamma}_{\widecheck n}.
    \]
    \item Le sottoreti associate alle due popolazioni sono assunte
    DAG connessi (\textit{Directed Acyclic Graphs}).
\end{itemize}

\end{frame}
