\begin{frame}{Tempi di percorrenza}

\begin{itemize}
    \item Ogni strada $r_h$ è caratterizzata da una funzione di costo

    \[
    \widehat{\tau}_h,
    \qquad
    \widecheck{\tau}_h.
    \]

    \item I costi dipendono dal traffico presente sulla rete.

    \item Il tempo necessario a percorrere un cammino è la somma
    dei costi delle strade che lo compongono:
\end{itemize}

\[
\widehat T
=
\widehat{\Gamma}^{\intercal}
\widehat{\tau},
\qquad
\widecheck T
=
\widecheck{\Gamma}^{\intercal}
\widecheck{\tau}.
\]

\vspace{0.3cm}

\[
\widehat T_M
=
\widehat\theta^{\intercal}\widehat T,
\qquad
\widecheck T_M
=
\widecheck\theta^{\intercal}\widecheck T.
\]

\end{frame}
