\begin{frame}{Distribuzione del traffico}

\begin{block}{Scelta dei percorsi}
I guidatori che scelgono ciascun percorso sono descritti da

\[
\widehat{\theta}\in S^{\widehat n},
\qquad
\widecheck{\theta}\in S^{\widecheck n},
\]
\end{block}

\begin{block}{Definizione dei percorsi}
Le matrici percorso-strada:

\[
\widehat{\Gamma},
\qquad
\widecheck{\Gamma}
\]

permettono di ottenere i guidatori sulle strade:

\[
\widehat{\nu}
=
\widehat{\Gamma}\widehat{\theta},
\qquad
\widecheck{\nu}
=
\widecheck{\Gamma}\widecheck{\theta}.
\]
\end{block}

\end{frame}
