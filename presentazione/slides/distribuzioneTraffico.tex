\begin{frame}{Distribuzione del traffico}

\begin{block}{Scelta dei percorsi}
Le quote di utenti che scelgono ciascun percorso sono descritte da

\[
\widehat{\theta}\in S^{\widehat n},
\qquad
\widecheck{\theta}\in S^{\widecheck n},
\]

dove $S^n$ è il simplesso delle probabilità.
\end{block}

\begin{block}{Flussi sulle strade}
Le matrici di incidenza percorso-strada

\[
\widehat{\Gamma},
\qquad
\widecheck{\Gamma}
\]

permettono di ottenere i flussi sulle strade:

\[
\widehat{\nu}
=
\widehat{\Gamma}\widehat{\theta},
\qquad
\widecheck{\nu}
=
\widecheck{\Gamma}\widecheck{\theta}.
\]
\end{block}

\end{frame}
